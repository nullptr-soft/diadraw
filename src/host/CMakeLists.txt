# Text files compiled in as one string each: <name>_data.cpp from src/<name>_data.cpp.in, whose
# @DIADRAW_<NAME>_TEXT@ is the file's text in a raw string literal delimited by `<delimiter>`.
function(diadraw_embed_text name file delimiter)
  string(TOUPPER "${name}" upper)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" text)
  string(FIND "${text}" ")${delimiter}\"" clash)
  if(NOT clash EQUAL -1)
    message(FATAL_ERROR "${file} must not contain )${delimiter}\"")
  endif()
  # MSVC refuses a string literal over 16 KB (C2026); say so here rather than in one compiler.
  string(LENGTH "${text}" length)
  if(length GREATER 16000)
    message(FATAL_ERROR "${file} is over 16000 bytes; split the generated literal")
  endif()
  set(DIADRAW_${upper}_TEXT "${text}")
  configure_file(src/${name}_data.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/${name}_data.cpp" @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/${name}_data.cpp" PROPERTIES
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_sources(diadraw_host PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${name}_data.cpp")
endfunction()

# Google Drive and OneDrive for the native shells (host/cloud.hpp), with the platform's HTTPS
# client: WinHTTP on Windows, libcurl elsewhere (part of macOS; libcurl4-openssl-dev on Linux).
# The OAuth clients are registered by whoever ships DiaDraw (docs/guide/cloud-storage.md); left
# empty, the service shows as not set up. Environment variables of the same names override them.
set(DIADRAW_GOOGLE_CLIENT_ID "" CACHE STRING "OAuth client ID of a Google Cloud \"Desktop app\" client")
set(DIADRAW_GOOGLE_CLIENT_SECRET "" CACHE STRING "The client secret Google issues with that Desktop app client")
set(DIADRAW_ONEDRIVE_CLIENT_ID "" CACHE STRING
    "Application (client) ID of a Microsoft Entra app with a \"Mobile and desktop\" platform")
# Without libcurl (outside Windows) there is no diadraw_cloud: the core, the CLI and the other tests
# still build, and the shells that need it say so.
if(NOT WIN32)
  find_package(CURL)
  find_package(Threads)
endif()
if(WIN32 OR (CURL_FOUND AND Threads_FOUND))
  add_library(diadraw_cloud STATIC src/cloud.cpp src/cloud_http.cpp src/cloud_loopback.cpp)
  target_link_libraries(diadraw_cloud PUBLIC diadraw_host PRIVATE diadraw_warnings)
  target_compile_definitions(diadraw_cloud PRIVATE
    DIADRAW_GOOGLE_CLIENT_ID="${DIADRAW_GOOGLE_CLIENT_ID}"
    DIADRAW_GOOGLE_CLIENT_SECRET="${DIADRAW_GOOGLE_CLIENT_SECRET}"
    DIADRAW_ONEDRIVE_CLIENT_ID="${DIADRAW_ONEDRIVE_CLIENT_ID}")
  if(WIN32)
    target_link_libraries(diadraw_cloud PRIVATE winhttp ws2_32)
  else()
    target_link_libraries(diadraw_cloud PRIVATE CURL::libcurl Threads::Threads)
  endif()
else()
  message(STATUS "libcurl not found: building without Google Drive and OneDrive (diadraw_cloud)")
endif()
