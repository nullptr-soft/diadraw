import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Ein Modell, zwei Editoren" icon="pencil">
    Definiere Elemente und Beziehungen einmal und bearbeite sie dann im Text oder auf der Zeichenfläche. Jede Änderung auf der Zeichenfläche wird als kleine Änderung in den Text zurückgeschrieben. So bleibt der Text maßgeblich und Änderungen sind in Git leicht nachzuvollziehen.
  </Card>
  <Card title="Viele Ansichten eines Modells" icon="seti:folder">
    Ein Systemkontext, die zugehörigen Container, eine Folge von Aufrufen: Jedes Diagramm ist eine Ansicht, die Teile desselben Modells auswählt und anordnet.
  </Card>
  <Card title="Ein Layout, das du steuern kannst" icon="setting">
    Schichten-, Baum-, Raster-, Radial-, kräftebasierte, feste und Sequenzlayouts. Wenn du einen Kasten ziehst, entsteht ein Hinweis wie `pin db below api` statt einer Koordinatenangabe. So bleibt das Layout übersichtlich, auch wenn das Modell wächst.
  </Card>
  <Card title="Eigene Diagramme mitbringen" icon="document">
    Importiere PlantUML, Mermaid, Structurizr, draw.io und BPMN. Exportiere SVG, PNG, PDF, draw.io und BPMN.
  </Card>
</CardGrid>
